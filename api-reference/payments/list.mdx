---
title: "Listar pagamentos"
api: "GET /v1/payments"
description: "Lista paginada (cursor) de pagamentos da tenant."
---

**Escopo:** `payments:read`

## Query params

<ParamField query="per_page" type="integer" default="25">1-100.</ParamField>
<ParamField query="cursor" type="string">Cursor.</ParamField>
<ParamField query="status" type="string">Filtra por status: `pending`, `processing`, `succeeded`, `failed`, `refunded`, `partially_refunded` ou `cancelled`.</ParamField>
<ParamField query="gateway" type="string">Filtra por gateway: `pagarme` (vendas em BRL), `stripe` (vendas internacionais) ou, em pagamentos antigos, `mercadopago` e `asaas`.</ParamField>
<ParamField query="user_id" type="integer">Filtra por usuário.</ParamField>

## Response

<ResponseField name="data[].id" type="integer">ID interno.</ResponseField>
<ResponseField name="data[].public_id" type="string">ID público (UUID).</ResponseField>
<ResponseField name="data[].user_id" type="integer">FK do user.</ResponseField>
<ResponseField name="data[].checkout_id" type="integer|null">FK do checkout.</ResponseField>
<ResponseField name="data[].subscription_id" type="integer|null">FK da assinatura.</ResponseField>
<ResponseField name="data[].gateway" type="string">`pagarme`, `stripe`, `mercadopago` ou `asaas`.</ResponseField>
<ResponseField name="data[].method" type="string">`card`, `card_installments`, `pix` ou `boleto`.</ResponseField>
<ResponseField name="data[].status" type="string">Mesmos valores do filtro `status`.</ResponseField>
<ResponseField name="data[].amount" type="integer">Valor em centavos.</ResponseField>
<ResponseField name="data[].original_amount" type="integer">Valor antes do desconto.</ResponseField>
<ResponseField name="data[].discount_amount" type="integer">Desconto aplicado.</ResponseField>
<ResponseField name="data[].gateway_fee" type="integer">Taxa do gateway.</ResponseField>
<ResponseField name="data[].net_amount" type="integer">Líquido.</ResponseField>
<ResponseField name="data[].currency" type="string">ISO 4217 (`BRL`, `USD`).</ResponseField>
<ResponseField name="data[].installments" type="integer">Parcelas.</ResponseField>
<ResponseField name="data[].created_at" type="string">ISO 8601.</ResponseField>
<ResponseField name="data[].updated_at" type="string">ISO 8601.</ResponseField>
